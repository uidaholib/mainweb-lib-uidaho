{% comment %} Simple line chart, drawn as SVG by "js/line-chart-js.html" with a hover tooltip. The data is always available in a "Show data as a table" toggle below the chart, which is also what shows if JavaScript is off. Add `custom-foot: js/line-chart-js.html` to the page front matter to use (it holds the chart styles too). e.g. --> {% include feature/line-chart.html caption="Cost per use by fiscal year" labels="FY23,FY24,FY25" series="Cost per use|clearwater|14.70,23.31,44.08" %} Options: - "caption" = chart title, shown above the chart. - "note" = optional, a short line of explanation shown under the caption. - "labels" = x-axis labels, comma separated, e.g. "FY23,FY24,FY25". - "series" = one or more lines, separated by semicolons. Each is "Name|color|values|short name", where color is a brand color name (e.g. clearwater, lupine, leather), values are comma separated numbers matching the labels, and short name is optional, used to label the end of the line on wide screens. A legend is added when there is more than one line. - "format" = "dollars" (default, $14.70) or "dollars-whole" ($1,450,000, with axis ticks like $1.5M). - "label_heading" = optional, the table heading for the labels column (default "Fiscal year"). - "size" = optional, "lg" for a taller chart. {%- endcomment -%} {%- assign lc_format = include.format | default: "dollars" -%} {%- assign lc_labels = include.labels | split: "," -%} {%- assign lc_series = include.series | split: ";" -%} {%- capture lc_json -%}[{%- for s in lc_series -%}{%- assign parts = s | split: "|" -%}{"name":{{ parts[0] | strip | jsonify }},"color":{{ parts[1] | strip | jsonify }},"values":[{{ parts[2] | strip }}],"short":{{ parts[3] | default: parts[0] | strip | jsonify }}}{% unless forloop.last %},{% endunless %}{%- endfor -%}]{%- endcapture -%}
